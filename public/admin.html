Sample JSON Script (Exact Format)

Follow this exact JSON structure to add or modify questions. Once uploaded, it automatically configures all test cases, starter codes & requirements! 

{
  "title": "House Robber",
  "description": "<p>You are a professional robber planning to rob houses along a street. Adjacent houses have security systems connected.</p><p>Return the maximum amount of money you can rob tonight without alerting the police.</p>",
  "difficulty": "Medium",
  "points": 100,
  "functionName": "rob",
  "constraints": [
    "1 <= nums.length <= 100",
    "0 <= nums[i] <= 400"
  ],
  "testCases": [
    {
      "input": { "nums": [1, 2, 3, 1] },
      "output": 4,
      "explanation": "Rob house 1 (money = 1) and house 3 (money = 3). Total = 4.",
      "isSample": true
    },
    {
      "input": { "nums": [2, 7, 9, 3, 1] },
      "output": 12,
      "explanation": "Rob house 1 (2), house 3 (9), house 5 (1). Total = 12.",
      "isSample": true
    },
    {
      "input": { "nums": [2, 1, 1, 2] },
      "output": 4,
      "isSample": false
    },
    {
      "input": { "nums": [0] },
      "output": 0,
      "isSample": false
    },
    {
      "input": { "nums": [5, 1, 1, 5] },
      "output": 10,
      "isSample": false
    }
  ]
}
